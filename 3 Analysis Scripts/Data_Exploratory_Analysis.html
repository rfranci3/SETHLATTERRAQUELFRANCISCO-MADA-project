Sampling period and MDR

summary_stats <- stork_AMR_clean %>%
  group_by(samp) %>%
  summarise(
    mean_MDR = mean(MDR),  # Proportion of MDR = 1
    se_MDR = sqrt(mean_MDR * (1 - mean_MDR) / n())  # Standard error of the proportion
  )

# Plot with box plot and error bars
pMDR <- stork_AMR_clean %>%
  ggplot(aes(x = as.factor(samp), y = MDR)) +
  geom_boxplot(alpha = 0.6, outlier.shape = NA) +  # Single box plot per samp
  geom_jitter(position = position_jitter(width = 0.2, height = 0), 
              alpha = 0.5, color = "black") +  # Add raw data points
  geom_errorbar(data = summary_stats,  # Error bars on top
                aes(x = as.factor(samp), 
                    ymin = mean_MDR - se_MDR, 
                    ymax = mean_MDR + se_MDR), 
                width = 0.2, color = "black", size = 0.8, inherit.aes = FALSE) +

create_and_save_heatmap <- function(data, year, filepath) {
  heatmap <- ggplot(data, aes(x = nes, y = Gene, fill = Abundance)) +
    geom_tile(na.rm = FALSE) + # Ensures NA values are kept
    facet_wrap(~samp, scales = "free", nrow = 1) +
    labs(title = paste("Gene Abundance across Sampling Periods for", year),
         x = "Nest ID", y = "Resistance Gene") +
    # Use a two-color gradient with NA values as white
    scale_fill_gradient(
      low = "dodgerblue2", 
      high = "red", 
      na.value = "white", # Set NA values to white
      name = "Abundance"
    ) +
    theme_minimal() +
    scale_x_discrete(guide = guide_axis(n.dodge = 2)) +
    theme(axis.text.x = element_text(angle = 90, vjust = 0.5, hjust = 1, size = 8))
  
  # Save the plot as a TIFF file
  ggsave(filename = filepath, plot = heatmap, device = "tiff", width = 10, height = 6, dpi = 300)
}
