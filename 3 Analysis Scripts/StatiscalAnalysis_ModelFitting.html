# Function to check variance of random effect
check_variance <- function(model, model_name) {
  variance <- summary(model)$varcor
  singular <- isSingular(model, tol = 1e-4)
  
  return(data.frame(
    Model = model_name,
    Variance = ifelse(length(variance) > 0, as.numeric(variance[[1]][1]), NA),
    Singular = singular
  ))
}
